//--- Statements (grammar.md §7)

void stmts() {
    // C-style for
    for (u32 i = 0; i < 10; i++) {
        continue;
    }

    // range-based for
    for (u32 j : range) {
        break;
    }

    // destructuring range-based for
    for (const u8 [ k, v ] : pairs) {
        var ignore = k;
    }

    // for with `var`
    for (var it : items) {
        continue;
    }

    // if / else, while, switch as expressions
    u32 r = if (x > 1) { 1 } else { 2 };

    while (x < 5) {
        x += 1;
    }

    loop {
        break;
    }

    switch (x) {
        Color::Red => { u32 a = 1; },
        Some { value } => { u32 b = value; },
        _ => { u32 c = 0; },
        else => { u32 d = 9; },
    }

    // defer / asm / return
    defer release();
    asm "nop" { outputs }
    return;
}

void exprs() {
    u32 a = b + c * d - e / f % g;
    bool t = x < y && y <= z || p != q && r == s;
    u8 m = bits | 1 ^ 2 & 3;
    u32 s = v << 2 >> 1;
    bool n = !flag;
    u32 nn = ~bits;
    i32 neg = -v;
    u8 deref = *ptr;
    rawptr addr = &value;
    ++counter;
    counter--;
    --other;

    // postfix
    u64 r1 = obj.field;
    u64 r2 = obj->field;
    u64 r3 = list[i];
    u64 r4 = obj.method(args);
    u64 r5 = ns::obj.method(args);
    u64 r6 = v as usize;

    // conditional
    u32 c = a > b ? a : b;

    // struct literal with `.field = x`
    var p = Point { .x = 1.0, .y = 2.0 };
    var opt = Type::Some { .field = x };

    // array / tuple literals
    u8[3] arr = [1, 2, 3];
    var tup = (1, 2, 3);

    // builtin identifiers
    usize sz = sizeof(u32);
    usize al = alignof(u64);
    u32 ty = typeof(x);
    assert(ok);
    assertOut(ok, msg);

    // compile-time attrs
    @isFormatLiteral
    u32 fmt = 1;
    @compileError("nope")
    u32 bad = 2;
    @assert(sz == 4)
    u32 good = 3;

    // assignment operators
    a += 1;
    a -= 1;
    a *= 2;
    a /= 2;
    a %= 2;
    a |= 1;
    a &= 1;
    a ^= 1;
    a <<= 1;
    a >>= 1;
}

//--- comptime semantics (grammar.md, observed in std)

// `comptime T` means T is deduced from the function's comptime block.
print<comptime T>(T fmt, usize len) {
    comptime {
        switch (fmt) {
            "u8" => { T = u8; },
            "u32" => { T = u32; },
        }
    }
}

u32 comptIf<comptime K>(K key) {
    comptime {
        for (u32 i = 0; i < 4; i++) {
            switch (i) {
                1 => { K = u8; },
                else => { K = u32; },
            }
        }
    }
    return 0;
}
